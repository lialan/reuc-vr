# The Wii common key for reading disc images in place (wii_disc.h) is the player's (common-key.bin next to the image). A build may be given one
# instead: UC_WII_COMMON_KEY_FILE names a 16-byte file on the build machine; its bytes go into this build only, never into the sources.
set(UC_WII_COMMON_KEY_FILE "" CACHE FILEPATH "16-byte Wii common key built into wii_disc.cpp (empty: none, the player supplies common-key.bin)")
if(UC_WII_COMMON_KEY_FILE)
  file(SIZE "${UC_WII_COMMON_KEY_FILE}" uc_key_size)
  if(NOT uc_key_size EQUAL 16)
    message(FATAL_ERROR "UC_WII_COMMON_KEY_FILE must hold exactly 16 bytes")
  endif()
  file(READ "${UC_WII_COMMON_KEY_FILE}" uc_key_hex HEX)
  set_source_files_properties(wii_disc.cpp PROPERTIES COMPILE_DEFINITIONS "UC_WII_COMMON_KEY_HEX=\"${uc_key_hex}\"")
  message(STATUS "Wii common key: built in from ${UC_WII_COMMON_KEY_FILE}")
endif()
