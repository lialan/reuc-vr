cmake_minimum_required(VERSION 3.25)
project(uc LANGUAGES C CXX)   # C for the third-party mod runtimes (miniz, Lua, QuickJS)
if(WIN32)
  enable_language(RC)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# clang-cl (Windows) takes MSVC-style options; Apple clang and the Android NDK take GNU-style ones.
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  set(UC_MSVC_FRONTEND ON)
else()
  set(UC_MSVC_FRONTEND OFF)
endif()
# Platform features without a macOS/Android path yet: off there by default.
option(UC_RENDERER_DIRECTX "build the DirectX 11/12 renderers" ${WIN32})
option(UC_VENDOR_UPSCALERS "build DLSS/Streamline/NGX and FSR (FidelityFX) support" ${WIN32})
option(UC_BUILD_LAUNCHER "build uc_launcher" ${WIN32})
option(UC_BUILD_UPDATER "build the isolated updater" ${WIN32})
option(UC_NATIVE_MODS "load native (DLL) mods" ${WIN32})
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND UC_MSVC_FRONTEND)
  add_compile_options(/clang:-march=x86-64-v3)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  # NEON is part of the arm64 baseline; the CPU target adds the newer instructions (fp16, dot product, LSE atomics, RCPC).
  # apple-m1 runs on every Apple Silicon Mac. Other arm64 targets (Android, Linux) pick their own CPU: there is no safe default above
  # the baseline (an M1-only instruction would be SIGILL on an older core).
  if(APPLE)
    set(UC_TARGET_CPU "apple-m1" CACHE STRING "-mcpu for arm64 builds")
  else()
    set(UC_TARGET_CPU "" CACHE STRING "-mcpu for arm64 builds (e.g. cortex-a78 for a Quest 3 build); empty = the arm64 baseline")
  endif()
  if(UC_TARGET_CPU)
    add_compile_options(-mcpu=${UC_TARGET_CPU})
  endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wno-deprecated-declarations)   # "unsafe CRT function" (getenv, fopen), winsock and std::filesystem::u8path notes in the tools and third-party code are noise
endif()
if(ANDROID)
  # The NDK's libc++ (LLVM 18) keeps std::jthread/stop_token behind -fexperimental-library (libc++experimental.a ships with the NDK).
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fexperimental-library>)
  add_link_options(-fexperimental-library)
endif()
# Profile-guided optimization: ~8% more frames in gameplay. `python scripts/pgo.py` records project/pgo/uc.profdata from a harness run of an
# instrumented build; while that file exists every build uses it (stale parts are ignored silently, re-record after big changes).
option(UC_PGO_GENERATE "build an instrumented uc.exe that records a profile (scripts/pgo.py)" OFF)
set(UC_PGO_PROFILE "${CMAKE_SOURCE_DIR}/project/pgo/uc.profdata" CACHE FILEPATH "profile for the optimizer; used when the file exists")
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND UC_MSVC_FRONTEND)
  if(UC_PGO_GENERATE)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/clang:-fprofile-instr-generate>)
    get_filename_component(uc_llvm_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    file(GLOB uc_profile_runtime "${uc_llvm_bin}/../lib/clang/*/lib/windows/clang_rt.profile-x86_64.lib")
    if(NOT uc_profile_runtime)
      message(FATAL_ERROR "clang_rt.profile-x86_64.lib not found next to ${CMAKE_CXX_COMPILER}")
    endif()
    add_link_options("${uc_profile_runtime}")
  elseif(EXISTS "${UC_PGO_PROFILE}")
    # A new profile must rebuild every object; mixed ThinLTO profile summaries cannot link.
    file(SHA256 "${UC_PGO_PROFILE}" uc_pgo_hash)
    set(uc_pgo_input "${CMAKE_BINARY_DIR}/pgo/${uc_pgo_hash}.profdata")
    configure_file("${UC_PGO_PROFILE}" "${uc_pgo_input}" COPYONLY)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/clang:-fprofile-instr-use=${uc_pgo_input}> -Wno-profile-instr-unprofiled -Wno-profile-instr-missing -Wno-profile-instr-out-of-date)
    message(STATUS "PGO profile: ${UC_PGO_PROFILE}")
  endif()
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  if(UC_PGO_GENERATE)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fprofile-instr-generate>)
    add_link_options(-fprofile-instr-generate)
  elseif(EXISTS "${UC_PGO_PROFILE}")
    file(SHA256 "${UC_PGO_PROFILE}" uc_pgo_hash)
    set(uc_pgo_input "${CMAKE_BINARY_DIR}/pgo/${uc_pgo_hash}.profdata")
    configure_file("${UC_PGO_PROFILE}" "${uc_pgo_input}" COPYONLY)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fprofile-instr-use=${uc_pgo_input}> -Wno-profile-instr-unprofiled -Wno-profile-instr-missing -Wno-profile-instr-out-of-date)
    message(STATUS "PGO profile: ${UC_PGO_PROFILE}")
  endif()
endif()
# ThinLTO across the translated game, runtime and renderer (lets the small native helpers be inlined into generated code); the link is slow.
option(UC_THINLTO "build with ThinLTO (build.bat and scripts/dev.bat turn it on)" OFF)
if(UC_THINLTO AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND UC_MSVC_FRONTEND)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/clang:-flto=thin>)
  add_link_options("/lldltocache:${CMAKE_BINARY_DIR}/lto-cache")
elseif(UC_THINLTO AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-flto=thin>)
  add_link_options(-flto=thin)
  if(APPLE)
    add_link_options("LINKER:-cache_path_lto,${CMAKE_BINARY_DIR}/lto-cache")
  else()
    add_link_options("LINKER:--thinlto-cache-dir=${CMAKE_BINARY_DIR}/lto-cache")
  endif()
endif()

if(UC_BUILD_UPDATER)
  # Portable clients and the isolated updater need the compiler's redistributable runtime.
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
  set(uc_helper_runtime_names "uc_updater.exe\n")
  set(uc_helper_runtime_sources "$<TARGET_FILE:uc_updater>\n")
  foreach(uc_runtime_file IN LISTS CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS)
    get_filename_component(uc_runtime_name "${uc_runtime_file}" NAME)
    string(APPEND uc_helper_runtime_names "${uc_runtime_name}\n")
    string(APPEND uc_helper_runtime_sources "${uc_runtime_file}\n")
  endforeach()
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/updater-runtime-files.txt" CONTENT "${uc_helper_runtime_names}")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/updater-runtime-sources.txt" CONTENT "${uc_helper_runtime_sources}")
endif()
